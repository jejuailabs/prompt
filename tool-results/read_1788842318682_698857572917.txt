    14→async function getZai() {
    15→  if (!zai) {
    16→    const mod = await import('z-ai-web-dev-sdk');
    17→    const ZAI = mod.default;
    18→    zai = await ZAI.create();
    19→  }

    21→}

    23→function svgPlaceholder(title: string, c1: string, c2: string, glyph: string): string {
    24→  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="768" height="768">
    25→  <defs><linearGradient id="g" x1="0" y1="0" x2="1" y2="1">
    26→    <stop offset="0" stop-color="${c1}"/><stop offset="1" stop-color="${c2}"/></linearGradient></defs>
    27→  <rect width="768" height="768" fill="url(#g)"/>
    28→  <circle cx="580" cy="180" r="140" fill="rgba(255,255,255,0.08)"/>
    29→  <circle cx="140" cy="620" r="190" fill="rgba(0,0,0,0.12)"/>
    30→  <text x="384" y="420" font-size="180" text-anchor="middle">${glyph}</text>
    31→  <text x="384" y="540" font-size="40" font-weight="700" text-anchor="middle" fill="rgba(255,255,255,0.92)" font-family="sans-serif">${title}</text>
    32→</svg>`;
    33→  return 'data:image/svg+xml;base64,' + Buffer.from(svg).toString('base64');
    34→}

    36→async function genImage(prompt: string, relPath: string, title: string, size = '1024x1024'): Promise<string> {
    37→  const outPath = path.join('public', relPath);
    38→  if (fs.existsSync(outPath)) {
    39→    console.log('• image exists, skip', relPath);
    40→    return '/' + relPath;
    41→  }
    42→  try {
    43→    const z = await getZai();
    44→    const res = await z.images.generations.create({ prompt, size });
    45→    const b64 = res?.data?.[0]?.base64;
    46→    if (!b64) throw new Error('empty response');
    47→    fs.mkdirSync(path.dirname(outPath), { recursive: true });
    48→    fs.writeFileSync(outPath, Buffer.from(b64, 'base64'));
    49→    console.log('✓ image', relPath);
    50→    return '/' + relPath;
    51→  } catch (e: any) {
    52→    console.error('✗ image failed, SVG fallback:', relPath, e?.message ?? e);
    53→    fs.mkdirSync(path.dirname(outPath), { recursive: true });
    54→    const svg = svgPlaceholder(title, '#7c3aed', '#2e1065', '✦');
    55→    fs.writeFileSync(outPath + '.svg', Buffer.from(svg.split(',')[1], 'base64'));
    56→    return '/' + relPath + '.svg';
    57→  }
    58→}
